/* ====== จัดการข้อมูลผู้ใช้ -> รหัส ณก14/05/40 ====== */
function saveUserInfo(){
  const fname = (document.getElementById("fname").value||"").trim();
  const lname = (document.getElementById("lname").value||"").trim();
  const bdate = document.getElementById("bdate").value;
  if(!fname||!lname||!bdate){ alert("กรุณากรอกชื่อ นามสกุล และวันเกิดให้ครบ"); return; }
  const d = new Date(bdate);

  const month = String(d.getMonth()+1).padStart(2,"0");
  const yearBE2 = String(d.getFullYear()+543).slice(-2);

  localStorage.setItem("userCode", userCode);
  renderUserCode();
}

/* ====== เริ่มทำแบบประเมิน ====== */
function startQuiz(type){
  if(!userCode){ alert("กรุณากรอกข้อมูลผู้ใช้ก่อนทำแบบประเมิน"); return; }
  currentQuiz = quizzes[type];
  currentQ = 0; totalScore = 0;
  resultEl.innerHTML=""; historyEl.classList.add("hidden"); historyEl.innerHTML="";
  showQuestion();
}
function showQuestion(){
  if(currentQ < currentQuiz.questions.length){
    const q = currentQuiz.questions[currentQ];
    quizEl.innerHTML = `<h3>${q}</h3><div class="options"></div>`;
    const holder = quizEl.querySelector(".options");
    currentQuiz.options.forEach((opt,i)=>{
      const b = document.createElement("button"); b.className="btn"; b.textContent = opt;
      b.onclick = ()=> selectAnswer(i); holder.appendChild(b);
    });
  }else{ showResult(); }
}
function selectAnswer(i){ totalScore += currentQuiz.scores[i]; currentQ++; showQuestion(); }

function showResult(){
  quizEl.innerHTML = "";
  const interp = currentQuiz.interpret(totalScore);
  const nowTH = new Date().toLocaleString("th-TH");
  resultEl.innerHTML = `
    <h3>ผลการประเมิน</h3>
    <div class="pill">ผู้ประเมิน: ${userCode}</div>
    <div class="pill">วันที่: ${nowTH}</div>
    <p><b>${currentQuiz.name}</b><br>คะแนนรวม: <b>${totalScore}</b><br>ผลแปลผล: <b>${interp}</b></p>
    <div class="grid two">
      <button class="btn" onclick="backToMenu()">กลับสู่เมนู</button>
      <button class="btn warn" onclick="showHistory()">ดูประวัติ</button>
    </div>
  `;
  saveResult(currentQuiz.name,totalScore,interp,nowTH);
}

/* ====== บันทึก/แสดงประวัติ ====== */
function saveResult(testName,score,interpretation,thDate){
  let history = JSON.parse(localStorage.getItem("mentalHealthHistory") || "[]");
  history.push({date: thDate, user:userCode, test:testName, score:score, result:interpretation});
  localStorage.setItem("mentalHealthHistory", JSON.stringify(history));
}
function showHistory(){
  quizEl.innerHTML=""; resultEl.innerHTML=""; historyEl.classList.remove("hidden");
  const hist = JSON.parse(localStorage.getItem("mentalHealthHistory") || "[]").slice().reverse();
  let html = `<h3>📊 ประวัติการประเมิน</h3>
  <div class="note">สามารถส่งออก PDF/PNG หรืออัปโหลดขึ้น Google Drive ได้จากที่นี่</div>`;
  if(hist.length===0){ html += "<p>ยังไม่มีประวัติ</p>"; }
  else{
    html += hist.map((h,i)=>`
      <div class="history-entry">
        <b>${i+1}.</b> รหัส: ${h.user} · วันที่: ${h.date}<br>
        แบบประเมิน: ${h.test}<br>
        คะแนน: ${h.score} · ผล: ${h.result}
      </div>`).join("");
    html += `
      <div class="grid two" style="margin-top:12px">
        <button class="btn" onclick="exportPDF()">📄 ส่งออก PDF</button>
        <button class="btn" onclick="exportPNG()">🖼️ ส่งออก PNG</button>
      </div>
      <div class="grid two" style="margin-top:8px">
        <button class="btn" onclick="uploadHistoryJSON()">⬆️ อัปโหลด JSON ไป Google Drive</button>
        <button class="btn danger" onclick="clearHistory()">🗑️ ล้างประวัติทั้งหมด</button>
      </div>
      <div class="center" style="margin-top:8px"><button class="btn secondary" onclick="backToMenu()">กลับสู่เมนู</button></div>
    `;
  }
  historyEl.innerHTML = html;
}
function clearHistory(){ localStorage.removeItem("mentalHealthHistory"); showHistory(); }
function backToMenu(){ quizEl.innerHTML=""; resultEl.innerHTML=""; historyEl.innerHTML=""; historyEl.classList.add("hidden"); }

/* ====== เชื่อม Google Drive (ใส่คีย์ภายหลัง) ====== */
const API_KEY   = "YOUR_API_KEY"; // <- ใส่ทีหลัง
const CLIENT_ID = "YOUR_CLIENT_ID.apps.googleusercontent.com"; // <- ใส่ทีหลัง
const SCOPE = "https://www.googleapis.com/auth/drive.file";
let gInited=false;

function initGapi(){
  return gapi.client.init({
    apiKey: API_KEY,
    clientId: CLIENT_ID,
    scope: SCOPE,
    discoveryDocs: ["https://www.googleapis.com/discovery/v1/apis/drive/v3/rest"]
  });
}
function googleSignIn(){
  gapi.load("client:auth2", ()=>{
    initGapi().then(()=>{
      gInited=true;
      gapi.auth2.getAuthInstance().signIn().then(()=> alert("เชื่อม Google Drive สำเร็จ"));
    }).catch(()=> alert("กรุณาใส่ API KEY/CLIENT ID ให้ถูกต้อง"));
  });
}
function uploadTextToDrive(text, fileName, mime){
  if(!gInited || !gapi.auth2.getAuthInstance().isSignedIn.get()){ alert("กรุณาเชื่อม Google Drive ก่อน"); return; }
  const boundary = "-------314159265358979323846"; const delimiter = "\r\n--"+boundary+"\r\n"; const close = "\r\n--"+boundary+"--";
  const meta = {name:fileName,mimeType:mime};
  const base64Data = btoa(unescape(encodeURIComponent(text)));
  const body = delimiter+'Content-Type: application/json\r\n\r\n'+JSON.stringify(meta)+delimiter+'Content-Type: '+mime+'\r\nContent-Transfer-Encoding: base64\r\n\r\n'+base64Data+close;
  gapi.client.request({
    path:"/upload/drive/v3/files", method:"POST", params:{uploadType:"multipart"},
    headers:{ "Content-Type":"multipart/related; boundary="+boundary }, body
  }).then(()=> alert("อัปโหลดไฟล์เรียบร้อย"));
}
function uploadBlobToDrive(base64, fileName, mime){
  if(!gInited || !gapi.auth2.getAuthInstance().isSignedIn.get()){ alert("กรุณาเชื่อม Google Drive ก่อน"); return; }
  const boundary="-------314159265358979323846"; const delimiter="\r\n--"+boundary+"\r\n"; const close="\r\n--"+boundary+"--";
  const meta={name:fileName,mimeType:mime};
  const body = delimiter+'Content-Type: application/json\r\n\r\n'+JSON.stringify(meta)+delimiter+'Content-Type: '+mime+'\r\nContent-Transfer-Encoding: base64\r\n\r\n'+base64+close;
  gapi.client.request({
    path:"/upload/drive/v3/files", method:"POST", params:{uploadType:"multipart"},
    headers:{ "Content-Type":"multipart/related; boundary="+boundary }, body
  }).then(()=> alert("อัปโหลดไฟล์เรียบร้อย"));
}
